Example-4: A Pod with the environment variables

docker --> mysql container ---> environment variable MYSQL_ROOT_PASSWORD

docker container run -d --name mysqldb \
              -e MYSQL_ROOT_PASSWORD=devops \
			  -e MYSQL_DATABASE=nop \
			  -e MYSQL_USER=devops \
			  -e MYSQL_PASSWORD=devops \
			  mysql:5.7
			 
yaml
-------------------------
---
apiVersion: v1
kind: Pod
metadata:
  name: mysqldb
spec:
  containers:
    - name: mysql
      image: mysql:5.7
      env:
        - name: MYSQL_ROOT_PASSWORD
          value: devops
        - name: MYSQL_DATABASE
          value: nop
        - name: MYSQL_USER
          value: devops
        - name: MYSQL_PASSWORD
          value: devops
