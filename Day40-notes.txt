openssl req -x509 -nodes -days 365 -newkey rsa:2048 -keyout app1.key -out app1.crt -subj "/CN=app1.haseeops.com/o=app1.haseebops.com -addext "subjectAltName = DNS:app1.haseebops.com"

-x509 ==> create a self signed cerfication

- nodes --> private key without passphrase

- days --> 365 --> valid for 1 year

- rsa:2048 ---> secure key size

- SAN ( SubjectAltName ) - must exactly match the hostname
