Multi containers in Pod , events & logs

---------------------------------------------------

apiVersion: v1
kind: Pod
metadata:
  name: multi-container
  labels:
    app: nginx
    type: webserver
spec:
  containers:
    - name: container-write
      image: alpine
      command: ["/bin/sh"]
      args: ["-c", "while true; do date >> /var/log/index.html; sleep 5;done"]
      volumeMounts:
        - name: shared-volume
          mountPath: /var/log

    - name: container-read
      image: nginx
      ports:
        - containerPort: 80
      volumeMounts:
        - name: shared-volume
          mountPath: /usr/share/nginx/html

  volumes:
    - name: shared-volume
      emptyDir: {}
--------------------------------------------------------------------

Refer - https://kubernetes.io/docs/tasks/access-application-cluster/communicate-containers-same-pod-shared-volume/
